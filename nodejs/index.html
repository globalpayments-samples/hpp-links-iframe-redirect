Production webhook handler

Always verify X-GP-Signature (HMAC-SHA256 with your webhook secret) before trusting any payload.
Node webhook.js 
// Production webhook handler
app.post('/webhook', express.raw({ type: '*/*' }), (req, res) => {
  const signature = req.headers['x-gp-signature'];
  const expected  = crypto
    .createHmac('sha256', process.env.GP_WEBHOOK_SECRET)
    .update(req.body)
    .digest('hex');

  if (signature !== expected) return res.status(401).send('Unauthorized');

  const event = JSON.parse(req.body.toString());
  switch (event.type) {
    case 'TRANSACTION_PREAUTHORIZED': handleAuth(event.content[0]); break;
    case 'TRANSACTION_CAPTURED':      handleCapture(event.content[0]); break;
  }
  res.status(200).send('OK');
});
